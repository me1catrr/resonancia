  \coordinate (A) at (0,0.8); \coordinate (B) at (1.6,1.8); \coordinate (C) at (1.8,-0.2);

  \foreach \n in {B,C,D,E} \node[nodo] at (\n) {\n};

  \draw[trazo, color=rAzul!80] (D') -- node[above, etiqueta] {\footnotesize 9,2} (E');
  \node[nodo, rectangle, draw=rVerde, fill=rVerde!20] at (A') {A};
  \foreach \n in {B,C,D,E} \node[nodo] at (\n') {\n};
  \node[etiqueta, color=rGris, align=center] at (10.6,-1.35) {\footnotesize $\mathbf W$: matriz de adyacencia ponderada};
