% Portada de J2.2: calendario con huecos, agrupación por hora y unión de tablas.
\documentclass[tikz,border=0pt]{standalone}
\usepackage{resonancia}
\begin{document}
\begin{tikzpicture}
  \clip[rounded corners=6pt] (0,0) rectangle (12,3.6);
  \shade[top color=rAzul!16, bottom color=white] (0,0) rectangle (12,3.6);
  % calendario: filas = días, columnas = horas; un hueco
  \foreach \i in {0,...,5} { \foreach \j in {0,...,11} {
    \pgfmathtruncatemacro{\hueco}{(\i==3 && \j>=4 && \j<=6) ? 1 : 0}
    \ifnum\hueco=1
      \fill[rNaranja!25, draw=rNaranja, line width=0.4pt] ({0.5+0.3*\j},{3.2-0.38*\i}) rectangle ({0.5+0.3*(\j+1)},{3.2-0.38*(\i+1)});
    \else
      \pgfmathtruncatemacro{\t}{20+4*abs(\j-6)}
      \fill[rAzul!\t, draw=white, line width=0.4pt] ({0.5+0.3*\j},{3.2-0.38*\i}) rectangle ({0.5+0.3*(\j+1)},{3.2-0.38*(\i+1)});
    \fi
  } }
  \node[etiqueta, color=rAzul] at (2.3,0.7) {\small días × horas};
  \draw[flecha] (4.4,1.9) -- (5.3,1.9);
  % curva diaria
  \draw[eje] (5.6,0.9) -- (8.7,0.9);
  \draw[analisis, grueso] plot[smooth, domain=5.7:8.6, samples=60] (\x, {1.5+0.9*sin((\x-5.7)*60-60)});
  \node[etiqueta, color=rVerde] at (7.15,0.55) {\small ciclo diario};
  \draw[flecha] (8.9,1.9) -- (9.5,1.9);
  % dos tablas unidas
  \draw[trazo, color=rVioleta, fill=white] (9.7,2.2) rectangle (11.5,3.2);
  \draw[trazo, color=rVioleta, fill=white] (9.7,0.6) rectangle (11.5,1.6);
  \foreach \y in {2.45,2.7,2.95} \draw[color=rGrisClaro, line width=1pt] (9.8,\y) -- (11.4,\y);
  \foreach \y in {0.85,1.1,1.35} \draw[color=rGrisClaro, line width=1pt] (9.8,\y) -- (11.4,\y);
  \draw[color=rVioleta, line width=1.2pt] (10.6,2.2) -- (10.6,1.6);
  \node[etiqueta, color=rVioleta] at (10.6,0.3) {\small \texttt{join}};
\end{tikzpicture}
\end{document}
