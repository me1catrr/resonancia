  \draw[decorate, decoration={brace, amplitude=5pt}, trazo, color=rVerde] ([yshift=3pt]adc.north west) -- ([yshift=3pt]reg.north east) node[midway, above=6pt, etiqueta, color=rVerde] {señal digital, $t=n/f_s$};
